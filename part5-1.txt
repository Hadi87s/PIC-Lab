void setup() {
  pinMode(7, INPUT);    
  pinMode(13, OUTPUT);  
  Serial.begin(9600);       
}

void loop() {
  int irState = digitalRead(7);  
  
  if (irState == HIGH) {
    digitalWrite(13, HIGH);  
    Serial.println("No Obstacle");
  } else {
    digitalWrite(13, LOW);   
    Serial.println("Obstacle Detected!");
  }

  delay(500); 
}
