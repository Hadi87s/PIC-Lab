#include <Keypad.h> 

const byte row    = 4;    
const byte column = 4;    

char keys[row][column] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte pin_rows[row] = {2, 3, 4, 5};  
byte pin_column[column] = {6, 7, 8, 9};  

Keypad keypad = Keypad(makeKeymap(keys), pin_rows, pin_column, row, column);

void setup() {
  Serial.begin(9600);  
  Serial.println("Press any key on the keypad");
}

void loop() {
  char key = keypad.getKey();  

  if (key) {  
    Serial.println(key);
  }
}
