#include <TM1637Display.h>

#define CLK 3
#define DIO 4

TM1637Display display= TM1637Display(CLK, DIO);

void setup() {
  display.setBrightness(7); // Max brightness
}

void loop() {
  for (int i = 0; i <= 9999; i++) {
    display.showNumberDec(i, false); // true = leading zeros off
    // Use smaller delay with consistent refresh
   delay(1); 
  }
}
